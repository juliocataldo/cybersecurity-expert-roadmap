PROMPT COMANDO WINDOWS

CMD (command prompt)

#Mostra o diretoria que estamos
echo %cd%
# visualizar o que tem de um diretorio
dir
#criar pasta
mkdir curso
#criar aqruivo
echo Julio > nomes.txt
#limpar tela
cls 
#visualizar conteudo
type meses.txt
#escrver sem sobrescrever
echo Gomes >> nomes.txt
#copiar arquivo
copy nomes.txt novo.txt
#mover arquivo
move novo.txt ../
#deletar arquivo
del nomes.txt
#ocultar arquivo
attrib +h nomes.txt
#desfazer comando oculto attrib
attirib -h *nome arquivo*
#HELP DO COMANDO
*comando* /?
#visu arqv ocultos
dir /a
#remover diretorio(pasta)
rmdir *nome_pasta*
#remover pasta com conteudo dentro
rmdir /s *nome_pasta*
##Buscar algum conteudo 
dir /S *nome.extensao*

#################################################
#mostra as conexoes
netstat -n 
#mostra tcp
netstat -p tcp
#processos em execucao
tasklist
#encerrar processo (passa o PID)
taskkill -pid 6928
#usuarios que existem no sistema
net user
#usuario com detalhes 
net user Administrator
#add user (comando + nome + senha + /add)
net user desec Admin@123 /add
#remover usuer
net user desec /delete
#trabalhar com rede.. 
net user
#Network configurations shortcut
ncpa.cpl

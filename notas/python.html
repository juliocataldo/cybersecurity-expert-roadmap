#!/usr/bin/python
import socket, sys

ip = sys.argv[1]
porta = int(sys.argv[2])

if (len.sys.argv) <=2:
	print "Modo de uso do programa: 127.0.0.1 8080"
else:
	meusocket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
	res = meusocket.connect_ex((ip, porta))
	if (res ==0):
